-- 1. Tabla: planes
CREATE TABLE planes (
    id SERIAL PRIMARY KEY,
    nombre_plan VARCHAR(100) NOT NULL,
    tipo_plan VARCHAR(20) NOT NULL 
        CHECK (tipo_plan IN ('residencial', 'empresarial', 'prepago', 'pospago')),
    velocidad VARCHAR(50) NOT NULL,
    precio DECIMAL(10,2) NOT NULL,
    descripcion TEXT,
    estado VARCHAR(10) DEFAULT 'activo' 
        CHECK (estado IN ('activo', 'inactivo')),
    fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 2. Tabla: usuarios
CREATE TABLE usuarios (
    id SERIAL PRIMARY KEY,
    username VARCHAR(100) NOT NULL UNIQUE, -- Añadido UNIQUE para proteger tu API en el Login
    password VARCHAR(255) NOT NULL,
    dni VARCHAR(15) NOT NULL UNIQUE, -- Añadido UNIQUE para evitar clientes duplicados
    nombre_completo VARCHAR(150) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE, -- Añadido UNIQUE para logins o recuperar contraseña
    telefono VARCHAR(20),
    fecha_nacimiento DATE,
    ubigeo CHAR(6),
    zona VARCHAR(100),
    rol VARCHAR(15) DEFAULT 'usuario' 
        CHECK (rol IN ('admin', 'usuario')),
    primer_login SMALLINT DEFAULT 1,
    fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 3. Tabla: consultas
CREATE TABLE consultas (
    id SERIAL PRIMARY KEY,
    dni VARCHAR(15) NOT NULL,
    nombre VARCHAR(150) NOT NULL,
    email VARCHAR(100) NOT NULL,
    telefono VARCHAR(20),
    motivo_consulta TEXT,
    estado VARCHAR(15) DEFAULT 'pendiente' 
        CHECK (estado IN ('pendiente', 'convertido', 'rechazado')),
    fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 4. Tabla: contratos
CREATE TABLE contratos (
    id SERIAL PRIMARY KEY,
    usuario_id INT NOT NULL, -- Unificado a usuario_id como preferiste
    plan_id INT NOT NULL,
    fecha_inicio DATE NOT NULL,
    estado_contrato VARCHAR(15) DEFAULT 'activo' 
        CHECK (estado_contrato IN ('activo', 'suspendido', 'finalizado')),
    CONSTRAINT fk_contratos_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    CONSTRAINT fk_contratos_plan
        FOREIGN KEY (plan_id)
        REFERENCES planes(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE
);

-- 5. Tabla: facturas
CREATE TABLE facturas (
    id SERIAL PRIMARY KEY,
    usuario_id INT NOT NULL, -- Unificado a usuario_id
    monto DECIMAL(10,2) NOT NULL,
    fecha_emision DATE NOT NULL,
    fecha_vencimiento DATE NOT NULL,
    estado VARCHAR(15) DEFAULT 'pendiente' 
        CHECK (estado IN ('pagado', 'pendiente', 'vencido')),
    fecha_pago TIMESTAMPTZ DEFAULT NULL,
    comprobante_pdf VARCHAR(255),
    CONSTRAINT fk_facturas_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- 6. Tabla: tickets
CREATE TABLE tickets (
    id SERIAL PRIMARY KEY,
    usuario_id INT NOT NULL,
    titulo VARCHAR(200) NOT NULL,
    descripcion TEXT,
    estado VARCHAR(15) DEFAULT 'abierto' 
        CHECK (estado IN ('abierto', 'en_proceso', 'resuelto', 'cerrado')),
    prioridad VARCHAR(10) DEFAULT 'media' 
        CHECK (prioridad IN ('baja', 'media', 'alta')),
    fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_tickets_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- 7. Tabla: comentarios_publicos
CREATE TABLE comentarios_publicos (
    id SERIAL PRIMARY KEY,
    usuario_id INT NOT NULL,
    -- nombre_usuario eliminado exitosamente para usar JOIN con la tabla usuarios
    comentario TEXT NOT NULL,
    calificacion SMALLINT NOT NULL 
        CHECK (calificacion >= 0 AND calificacion <= 5), -- Acotado de 0 a 5 estrellas
    estado VARCHAR(15) DEFAULT 'pendiente' 
        CHECK (estado IN ('pendiente', 'aprobado', 'rechazado')),
    fecha_comentario TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_comentarios_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- 8. Tabla: mensajes_tickets
CREATE TABLE mensajes_tickets (
    id SERIAL PRIMARY KEY,
    ticket_id INT NOT NULL,
    usuario_id INT NOT NULL,
    mensaje TEXT NOT NULL,
    fecha_envio TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_mensajes_ticket
        FOREIGN KEY (ticket_id)
        REFERENCES tickets(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    CONSTRAINT fk_mensajes_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);
